## Usage

```luau
--!strict
export type Resolvable<Entries> = keyof<Entries>

export type Registry<Path> = {
    read Get: (self: Registry<Path>, path: Path) -> (unknown, boolean),
}

local function register<Entries>(entries: Entries): Registry<Resolvable<Entries>>
    const values = entries :: unknown :: { [string]: unknown }

    local function get(_, path: Resolvable<Entries>): (unknown, boolean)
        const value = values[path :: unknown :: string]
        if value == nil then
            return `Unknown path: {path}`, false
        end
        return value, true
    end

    return { Get = get }
end
```
