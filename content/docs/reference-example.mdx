## Usage

```luau
--!strict
export type Ipsum<Entries> = keyof<Entries>

export type Lorem<Path> = {
    read Get: (self: Lorem<Path>, path: Path) -> (unknown, boolean),
}

local function create<Entries>(entries: Entries): Lorem<Ipsum<Entries>>
    const values = entries :: unknown :: { [string]: unknown }

    local function get(_, path: Ipsum<Entries>): (unknown, boolean)
        const value = values[path :: unknown :: string]
        if value == nil then
            return `Unknown path: {path}`, false
        end
        return value, true
    end

    return { Get = get }
end
```

  <h5>Anchor a part after a tween</h5>

A [BasePart](https://create.roblox.com/docs/reference/engine/classes/BasePart) can be frozen after a tween finishes. See [BasePart.Anchored](https://create.roblox.com/docs/reference/engine/classes/BasePart#Anchored) for how anchoring affects physics.

```luau
local TweenService = game:GetService("TweenService")
local part = script.Parent
assert(part:IsA("BasePart"), "Put this script under a BasePart")

local tween = TweenService:Create(part, TweenInfo.new(1), {
    Transparency = 0.5,
})

tween.Completed:Connect(function(state)
    if state == Enum.PlaybackState.Completed then
        part.Anchored = true
    end
end)

tween:Play()
```

<h5>Track parts touching a sensor</h5>

Attach this script to a part with touch events enabled. It tracks the parts currently touching the sensor and updates a count attribute that another script can read.

```luau
local sensor = script.Parent
assert(sensor:IsA("BasePart"), "Put this script under a BasePart")

local touching: { [BasePart]: boolean } = {}

local function publishCount()
    local count = 0

    for _ in touching do
        count += 1
    end

    sensor:SetAttribute("TouchCount", count)
end

sensor.Touched:Connect(function(part)
    if touching[part] then
        return
    end

    touching[part] = true
    publishCount()
end)

sensor.TouchEnded:Connect(function(part)
    if not touching[part] then
        return
    end

    touching[part] = nil
    publishCount()
end)

publishCount()
```
