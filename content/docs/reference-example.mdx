## Usage

```luau
--!strict
export type Ipsum<Entries> = keyof<Entries>

export type Lorem<Path> = {
    read Get: (self: Lorem<Path>, path: Path) -> (unknown, boolean),
}

local function create<Entries>(entries: Entries): Lorem<Ipsum<Entries>>
    const values = entries :: unknown :: { [string]: unknown }

    local function get(_, path: Ipsum<Entries>): (unknown, boolean)
        const value = values[path :: unknown :: string]
        if value == nil then
            return `Unknown path: {path}`, false
        end
        return value, true
    end

    return { Get = get }
end
```
